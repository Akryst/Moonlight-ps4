cmake_minimum_required(VERSION 3.16)
project(moonlight_ui_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(PLATFORM_LINUX ON CACHE BOOL "Desktop target" FORCE)
include(FetchContent)
FetchContent_Declare(cross2d
    GIT_REPOSITORY https://github.com/Cpasjuste/libcross2d.git
    GIT_TAG 9dfe2fcb4ccb2dd7be04718ee7ddb450abd85de1)
FetchContent_MakeAvailable(cross2d)
find_package(PNG REQUIRED)
add_executable(moonlight-ui-preview main.cpp)
target_link_libraries(moonlight-ui-preview PRIVATE cross2d PNG::PNG)
target_compile_options(moonlight-ui-preview PRIVATE -Wall)
file(COPY assets DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
file(COPY ../pkg/assets/ui/buttons DESTINATION ${CMAKE_CURRENT_BINARY_DIR}/assets)
target_compile_definitions(moonlight-ui-preview PRIVATE PREVIEW_ASSETS="${CMAKE_CURRENT_BINARY_DIR}/assets")
enable_testing()
add_executable(preview-state-check state_check.cpp)
target_compile_options(preview-state-check PRIVATE -UNDEBUG)
add_test(NAME preview-navigation COMMAND preview-state-check)
add_executable(preview-catalog-check catalog_check.cpp)
target_link_libraries(preview-catalog-check PRIVATE config)
target_compile_options(preview-catalog-check PRIVATE -UNDEBUG)
add_test(NAME preview-catalog COMMAND preview-catalog-check)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME preview-catalog-export COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/catalog_check.py)
add_executable(preview-settings-check settings_check.cpp)
target_link_libraries(preview-settings-check PRIVATE cross2d)
target_compile_options(preview-settings-check PRIVATE -UNDEBUG)
add_test(NAME preview-settings-and-projection COMMAND preview-settings-check)
add_executable(preview-assets-check assets_check.cpp)
target_link_libraries(preview-assets-check PRIVATE PNG::PNG)
target_compile_options(preview-assets-check PRIVATE -UNDEBUG)
add_test(NAME preview-controller-images COMMAND preview-assets-check ${CMAKE_CURRENT_BINARY_DIR}/assets)
add_executable(native-ui-check native_ui_check.c ../src/ui/ui_draw.c)
target_compile_options(native-ui-check PRIVATE -UNDEBUG)
add_test(NAME native-ui-transitions-and-alpha COMMAND native-ui-check)
